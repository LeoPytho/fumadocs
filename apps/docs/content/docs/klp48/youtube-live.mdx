## What is YouTube Live API?

The YouTube Live API from @jkt48/core allows you to retrieve real-time information about JKT48V YouTube live streams that are currently active. This endpoint provides comprehensive live streaming data including stream details, member information, viewer counts, and platform-specific metadata for ongoing YouTube streams.

- Currently active YouTube live streams
- Real-time viewer counts and engagement metrics
- Member profile information with images
- Stream titles and descriptions
- Live chat and interaction data
- Platform-specific YouTube metadata

<Callout>
This API endpoint returns data about currently active YouTube live streams, making it perfect for building live stream monitoring applications and real-time dashboards.

All features require an API key for authentication and access.
</Callout>

## Advanced Examples

### Live Stream Monitor

```javascript
const jkt48Api = require('@jkt48/core');

class YouTubeLiveMonitor {
  constructor(apiKey, checkInterval = 30000) {
    this.apiKey = apiKey;
    this.checkInterval = checkInterval;
    this.currentStreams = new Map();
    this.isMonitoring = false;
  }
  
  async checkLiveStreams() {
    try {
      const liveStreams = await jkt48Api.liveYoutube(this.apiKey);
      
      // Check for new streams
      liveStreams.forEach(stream => {
        if (!this.currentStreams.has(stream.room_id)) {
          this.onStreamStarted(stream);
          this.currentStreams.set(stream.room_id, stream);
        } else {
          // Update existing stream data
          const existing = this.currentStreams.get(stream.room_id);
          if (existing.viewer_count !== stream.viewer_count) {
            this.onViewerCountChanged(stream, existing.viewer_count);
          }
          this.currentStreams.set(stream.room_id, stream);
        }
      });
      
      // Check for ended streams
      const currentRoomIds = new Set(liveStreams.map(s => s.room_id));
      for (const [roomId, stream] of this.currentStreams) {
        if (!currentRoomIds.has(roomId)) {
          this.onStreamEnded(stream);
          this.currentStreams.delete(roomId);
        }
      }
      
    } catch (error) {
      console.error('Error checking live streams:', error.message);
    }
  }
  
  onStreamStarted(stream) {
    console.log(`🔴 ${stream.name} started streaming: ${stream.title}`);
    console.log(`   Viewers: ${stream.viewer_count.toLocaleString()}`);
  }
  
  onStreamEnded(stream) {
    const duration = Math.floor(stream.duration / 60);
    console.log(`⏹️ ${stream.name} ended stream after ${duration} minutes`);
  }
  
  onViewerCountChanged(stream, previousCount) {
    const change = stream.viewer_count - previousCount;
    const direction = change > 0 ? '📈' : '📉';
    console.log(`${direction} ${stream.name}: ${previousCount.toLocaleString()} → ${stream.viewer_count.toLocaleString()}`);
  }
  
  startMonitoring() {
    if (this.isMonitoring) return;
    
    this.isMonitoring = true;
    console.log('🎬 Starting YouTube Live Monitor...');
    
    // Initial check
    this.checkLiveStreams();
    
    // Set up periodic checks
    this.monitorInterval = setInterval(() => {
      this.checkLiveStreams();
    }, this.checkInterval);
  }
  
  stopMonitoring() {
    if (!this.isMonitoring) return;
    
    this.isMonitoring = false;
    clearInterval(this.monitorInterval);
    console.log('⏹️ YouTube Live Monitor stopped');
  }
}

// Usage
const monitor = new YouTubeLiveMonitor('your-api-key-here', 30000);
monitor.startMonitoring();
```

### Stream Alert System

```javascript
async function createStreamAlertSystem() {
  const apiKey = 'your-api-key-here';
  let knownStreams = new Set();
  
  async function checkForNewStreams() {
    try {
      const currentStreams = await jkt48Api.liveYoutube(apiKey);
      
      currentStreams.forEach(stream => {
        if (!knownStreams.has(stream.room_id)) {
          sendStreamAlert(stream);
          knownStreams.add(stream.room_id);
        }
      });
      
      // Clean up ended streams
      if (currentStreams.length === 0) {
        knownStreams.clear();
      }
      
    } catch (error) {
      console.error('Error checking for new streams:', error.message);
    }
  }
  
  function sendStreamAlert(stream) {
    console.log('🚨 NEW LIVE STREAM ALERT!');
    console.log('========================');
    console.log(`Member: ${stream.name}`);
    console.log(`Title: ${stream.title}`);
    console.log(`Started: ${new Date(stream.started_at).toLocaleString()}`);
    console.log(`Current Viewers: ${stream.viewer_count.toLocaleString()}`);
    console.log(`Watch: ${stream.stream_url}`);
    console.log('========================\n');
    
    // Here you could integrate with notification services:
    // - Send email notifications
    // - Push notifications to mobile apps
    // - Discord/Slack webhooks
    // - SMS alerts
  }
  
  // Check every 15 seconds for new streams
  setInterval(checkForNewStreams, 15000);
  
  // Initial check
  checkForNewStreams();
}

// Start the alert system
createStreamAlertSystem();
```

### Live Dashboard Data

```javascript
async function getLiveDashboardData() {
  try {
    const apiKey = 'your-api-key-here';
    const liveStreams = await jkt48Api.liveYoutube(apiKey);
    
    if (liveStreams.length === 0) {
      console.log('📺 No YouTube streams currently live');
      return { active: false, streams: [] };
    }
    
    // Calculate dashboard statistics
    const dashboardData = {
      active: true,
      totalStreams: liveStreams.length,
      totalViewers: liveStreams.reduce((sum, stream) => sum + stream.viewer_count, 0),
      averageViewers: 0,
      topStream: null,
      memberCount: new Set(liveStreams.map(s => s.name)).size,
      streams: liveStreams.map(stream => ({
        member: stream.name,
        title: stream.title,
        viewers: stream.viewer_count,
        duration: Math.floor(stream.duration / 60),
        thumbnail: stream.thumbnail_url,
        url: stream.stream_url,
        chatEnabled: stream.chat_enabled
      }))
    };
    
    // Calculate averages and find top stream
    dashboardData.averageViewers = Math.round(dashboardData.totalViewers / dashboardData.totalStreams);
    dashboardData.topStream = liveStreams.reduce((top, stream) => 
      stream.viewer_count > (top?.viewer_count || 0) ? stream : top, null
    );
    
    console.log('📊 YouTube Live Dashboard');
    console.log('========================');
    console.log(`Active Streams: ${dashboardData.totalStreams}`);
    console.log(`Total Viewers: ${dashboardData.totalViewers.toLocaleString()}`);
    console.log(`Average Viewers: ${dashboardData.averageViewers.toLocaleString()}`);
    console.log(`Active Members: ${dashboardData.memberCount}`);
    
    if (dashboardData.topStream) {
      console.log(`\n🔝 Top Stream:`);
      console.log(`   ${dashboardData.topStream.name}: ${dashboardData.topStream.viewer_count.toLocaleString()} viewers`);
      console.log(`   "${dashboardData.topStream.title}"`);
    }
    
    console.log('\n📋 All Active Streams:');
    dashboardData.streams
      .sort((a, b) => b.viewers - a.viewers)
      .forEach((stream, index) => {
        console.log(`${index + 1}. ${stream.member}`);
        console.log(`   👁️ ${stream.viewers.toLocaleString()} viewers • ⏱️ ${stream.duration}m`);
        console.log(`   📺 "${stream.title}"`);
        console.log('');
      });
    
    return dashboardData;
    
  } catch (error) {
    console.error('Error fetching dashboard data:', error.message);
    return { active: false, streams: [], error: error.message };
  }
}
```

### Stream Quality Monitor

```javascript
class StreamQualityMonitor {
  constructor(apiKey) {
    this.apiKey = apiKey;
    this.streamHistory = new Map();
  }
  
  async monitorStreamQuality() {
    try {
      const liveStreams = await jkt48Api.liveYoutube(this.apiKey);
      
      liveStreams.forEach(stream => {
        if (!this.streamHistory.has(stream.room_id)) {
          this.streamHistory.set(stream.room_id, {
            startTime: Date.now(),
            viewerHistory: [stream.viewer_count],
            peakViewers: stream.viewer_count,
            averageViewers: stream.viewer_count,
            streamData: stream
          });
        } else {
          const history = this.streamHistory.get(stream.room_id);
          history.viewerHistory.push(stream.viewer_count);
          history.peakViewers = Math.max(history.peakViewers, stream.viewer_count);
          history.averageViewers = Math.round(
            history.viewerHistory.reduce((sum, count) => sum + count, 0) / history.viewerHistory.length
          );
          
          // Keep only last 50 viewer count records
          if (history.viewerHistory.length > 50) {
            history.viewerHistory = history.viewerHistory.slice(-50);
          }
        }
        
        this.analyzeStreamHealth(stream);
      });
      
      // Clean up ended streams
      const activeRoomIds = new Set(liveStreams.map(s => s.room_id));
      for (const roomId of this.streamHistory.keys()) {
        if (!activeRoomIds.has(roomId)) {
          this.generateStreamReport(roomId);
          this.streamHistory.delete(roomId);
        }
      }
      
    } catch (error) {
      console.error('Error monitoring stream quality:', error.message);
    }
  }
  
  analyzeStreamHealth(stream) {
    const history = this.streamHistory.get(stream.room_id);
    const recentViewers = history.viewerHistory.slice(-10);
    
    if (recentViewers.length >= 5) {
      const trend = this.calculateTrend(recentViewers);
      const stability = this.calculateStability(recentViewers);
      
      if (trend < -0.2) {
        console.log(`⚠️ ${stream.name}: Declining viewership trend`);
      }
      
      if (stability > 0.3) {
        console.log(`📊 ${stream.name}: High viewer volatility`);
      }
      
      if (stream.viewer_count < history.averageViewers * 0.5) {
        console.log(`📉 ${stream.name}: Viewers dropped significantly`);
      }
    }
  }
  
  calculateTrend(values) {
    if (values.length < 2) return 0;
    
    const first = values.slice(0, Math.floor(values.length / 2));
    const second = values.slice(Math.floor(values.length / 2));
    
    const firstAvg = first.reduce((sum, val) => sum + val, 0) / first.length;
    const secondAvg = second.reduce((sum, val) => sum + val, 0) / second.length;
    
    return (secondAvg - firstAvg) / firstAvg;
  }
  
  calculateStability(values) {
    if (values.length < 2) return 0;
    
    const mean = values.reduce((sum, val) => sum + val, 0) / values.length;
    const variance = values.reduce((sum, val) => sum + Math.pow(val - mean, 2), 0) / values.length;
    const standardDeviation = Math.sqrt(variance);
    
    return standardDeviation / mean; // Coefficient of variation
  }
  
  generateStreamReport(roomId) {
    const history = this.streamHistory.get(roomId);
    if (!history) return;
    
    const streamDuration = Math.floor((Date.now() - history.startTime) / (1000 * 60));
    
    console.log(`📈 Stream Report: ${history.streamData.name}`);
    console.log('====================================');
    console.log(`Duration: ${streamDuration} minutes`);
    console.log(`Peak Viewers: ${history.peakViewers.toLocaleString()}`);
    console.log(`Average Viewers: ${history.averageViewers.toLocaleString()}`);
    console.log(`Final Viewers: ${history.viewerHistory[history.viewerHistory.length - 1].toLocaleString()}`);
    
    const engagement = (history.averageViewers / history.peakViewers) * 100;
    console.log(`Engagement Rate: ${engagement.toFixed(1)}%`);
    console.log('====================================\n');
  }
  
  startMonitoring(interval = 60000) {
    console.log('📊 Starting Stream Quality Monitor...');
    
    // Initial check
    this.monitorStreamQuality();
    
    // Set up periodic monitoring
    this.monitorInterval = setInterval(() => {
      this.monitorStreamQuality();
    }, interval);
  }
  
  stopMonitoring() {
    if (this.monitorInterval) {
      clearInterval(this.monitorInterval);
      console.log('⏹️ Stream Quality Monitor stopped');
    }
  }
}

// Usage
const qualityMonitor = new StreamQualityMonitor('your-api-key-here');
qualityMonitor.startMonitoring(30000); // Check every 30 seconds
```

### Simple Stream Checker

```javascript
async function checkYoutubeLiveStatus() {
  try {
    const apiKey = 'your-api-key-here';
    const liveStreams = await jkt48Api.liveYoutube(apiKey);
    
    if (liveStreams.length === 0) {
      console.log('❌ No JKT48V members are currently live on YouTube');
      return false;
    }
    
    console.log(`✅ ${liveStreams.length} member${liveStreams.length > 1 ? 's' : ''} currently live on YouTube:`);
    
    liveStreams.forEach(stream => {
      const duration = Math.floor(stream.duration / 60);
      console.log(`🔴 ${stream.name} - ${stream.viewer_count.toLocaleString()} viewers (${duration}m)`);
    });
    
    return true;
    
  } catch (error) {
    console.error('Error checking YouTube live status:', error.message);
    return false;
  }
}

// Check live status every 5 minutes
setInterval(checkYoutubeLiveStatus, 5 * 60 * 1000);
```
