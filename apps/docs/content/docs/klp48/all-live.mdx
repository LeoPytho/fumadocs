## What is Live All Platforms API?

The Live All Platforms API from @jkt48/core allows you to retrieve real-time information about currently active JKT48 live streams across all supported platforms including YouTube, IDN Live, and SHOWROOM. This endpoint provides comprehensive live streaming data with direct streaming URLs, chat room information, and member details for all ongoing streams in a single API call.

- Real-time active streams from all platforms (YouTube, IDN, SHOWROOM)
- Direct streaming URLs with quality options
- Chat room integration details
- Member profile information with images
- Stream start timestamps and duration
- Platform-specific metadata and room IDs

<Callout>
This API endpoint returns real-time data about currently active live streams across all platforms, making it perfect for building live stream aggregators and monitoring dashboards.

All features require an API key for authentication and access.
</Callout>

## Data Field Explanations

<Accordions type="single">

<Accordion title="Basic Stream Information">
- **name**: Member name or stream title
- **img**: Stream thumbnail or member profile image
- **img_alt**: Alternative or higher quality image
- **url_key**: Platform-specific username or identifier
- **slug**: URL slug for the stream
- **type**: Platform type ("youtube", "idn", "showroom")
</Accordion>

<Accordion title="Platform & Room Data">
- **room_id**: Platform-specific room identifier
- **chat_room_id**: Chat room identifier (AWS IVS for IDN)
- **is_graduate**: Member graduation status
- **is_group**: Whether it's a group stream
</Accordion>

<Accordion title="Streaming Data">
- **started_at**: Stream start timestamp (ISO 8601)
- **streaming_url_list**: Array of streaming URLs with quality options
- **streaming_url_list[].label**: Quality label (e.g., "original", "720p", "480p")
- **streaming_url_list[].quality**: Quality level (1 = highest)
- **streaming_url_list[].url**: Direct streaming URL (HLS/m3u8)
</Accordion>

</Accordions>

## Advanced Examples

### Filter Streams by Platform

```javascript
const jkt48Api = require('@jkt48/core');

async function getStreamsByPlatform(platform) {
  try {
    const apiKey = 'your-api-key-here';
    const live = await jkt48Api.live(apiKey);
    
    const filteredStreams = live.filter(stream => stream.type === platform);
    
    console.log(`Current ${platform.toUpperCase()} streams:`);
    filteredStreams.forEach(stream => {
      const duration = Math.floor((Date.now() - new Date(stream.started_at)) / 60000);
      console.log(`${stream.name}: Live for ${duration} minutes`);
    });
    
    return filteredStreams;
  } catch (error) {
    console.error('Error filtering streams:', error.message);
  }
}

// Usage
getStreamsByPlatform('idn');
getStreamsByPlatform('showroom');
getStreamsByPlatform('youtube');
```

### Real-time Stream Monitor

```javascript
async function monitorLiveStreams() {
  try {
    const apiKey = 'your-api-key-here';
    let previousStreams = [];
    
    setInterval(async () => {
      const currentStreams = await jkt48Api.live(apiKey);
      
      // Check for new streams
      const newStreams = currentStreams.filter(current => 
        !previousStreams.some(prev => prev.room_id === current.room_id && prev.type === current.type)
      );
      
      // Check for ended streams
      const endedStreams = previousStreams.filter(prev => 
        !currentStreams.some(current => current.room_id === prev.room_id && current.type === prev.type)
      );
      
      if (newStreams.length > 0) {
        console.log('\n🟢 New streams started:');
        newStreams.forEach(stream => {
          console.log(`${stream.name} started streaming on ${stream.type.toUpperCase()}`);
        });
      }
      
      if (endedStreams.length > 0) {
        console.log('\n🔴 Streams ended:');
        endedStreams.forEach(stream => {
          console.log(`${stream.name} ended stream on ${stream.type.toUpperCase()}`);
        });
      }
      
      if (currentStreams.length > 0) {
        console.log(`\n📺 Currently live: ${currentStreams.length} streams`);
        currentStreams.forEach(stream => {
          const duration = Math.floor((Date.now() - new Date(stream.started_at)) / 60000);
          console.log(`  ${stream.name} (${stream.type}) - ${duration}m`);
        });
      }
      
      previousStreams = [...currentStreams];
    }, 30000); // Check every 30 seconds
    
  } catch (error) {
    console.error('Error monitoring streams:', error.message);
  }
}

// Start monitoring
monitorLiveStreams();
```

### Stream Aggregator Dashboard

```javascript
async function createStreamDashboard() {
  try {
    const apiKey = 'your-api-key-here';
    const live = await jkt48Api.live(apiKey);
    
    // Group by platform
    const platformGroups = live.reduce((groups, stream) => {
      if (!groups[stream.type]) groups[stream.type] = [];
      groups[stream.type].push(stream);
      return groups;
    }, {});
    
    console.log('🎬 JKT48 Live Stream Dashboard');
    console.log('===============================');
    
    Object.entries(platformGroups).forEach(([platform, streams]) => {
      console.log(`\n📱 ${platform.toUpperCase()} (${streams.length} active)`);
      
      streams.forEach(stream => {
        const startTime = new Date(stream.started_at);
        const duration = Math.floor((Date.now() - startTime) / 60000);
        const streamUrls = stream.streaming_url_list.length;
        
        console.log(`  ├─ ${stream.name}`);
        console.log(`  │  ⏱️  Live for ${duration} minutes`);
        console.log(`  │  🔗 ${streamUrls} quality option${streamUrls > 1 ? 's' : ''}`);
        console.log(`  │  💬 Chat: ${stream.chat_room_id ? 'Available' : 'N/A'}`);
      });
    });
    
    if (live.length === 0) {
      console.log('\n😴 No active streams at the moment');
    }
    
  } catch (error) {
    console.error('Error creating dashboard:', error.message);
  }
}
```

### Extract Streaming URLs

```javascript
async function getStreamingUrls(memberName = null) {
  try {
    const apiKey = 'your-api-key-here';
    const live = await jkt48Api.live(apiKey);
    
    let filteredStreams = live;
    if (memberName) {
      filteredStreams = live.filter(stream => 
        stream.name.toLowerCase().includes(memberName.toLowerCase())
      );
    }
    
    console.log('🔗 Available Streaming URLs:');
    
    filteredStreams.forEach(stream => {
      console.log(`\n${stream.name} (${stream.type.toUpperCase()})`);
      
      stream.streaming_url_list.forEach(url => {
        console.log(`  Quality: ${url.label} (${url.quality})`);
        console.log(`  URL: ${url.url}`);
      });
      
      if (stream.chat_room_id) {
        console.log(`  Chat Room: ${stream.chat_room_id}`);
      }
    });
    
    return filteredStreams.map(stream => ({
      member: stream.name,
      platform: stream.type,
      urls: stream.streaming_url_list
    }));
    
  } catch (error) {
    console.error('Error getting streaming URLs:', error.message);
  }
}

// Usage
getStreamingUrls(); // All streams
getStreamingUrls('Virgi'); // Specific member
```

### Platform Statistics

```javascript
async function getPlatformStats() {
  try {
    const apiKey = 'your-api-key-here';
    const live = await jkt48Api.live(apiKey);
    
    const stats = {
      total: live.length,
      platforms: {},
      members: new Set(),
      averageDuration: 0
    };
    
    let totalDuration = 0;
    
    live.forEach(stream => {
      // Platform counts
      if (!stats.platforms[stream.type]) {
        stats.platforms[stream.type] = 0;
      }
      stats.platforms[stream.type]++;
      
      // Unique members
      stats.members.add(stream.name);
      
      // Duration calculation
      const duration = Math.floor((Date.now() - new Date(stream.started_at)) / 60000);
      totalDuration += duration;
    });
    
    stats.averageDuration = live.length > 0 ? Math.round(totalDuration / live.length) : 0;
    stats.uniqueMembers = stats.members.size;
    
    console.log('📊 Live Stream Statistics');
    console.log('========================');
    console.log(`Total Active Streams: ${stats.total}`);
    console.log(`Unique Members: ${stats.uniqueMembers}`);
    console.log(`Average Duration: ${stats.averageDuration} minutes`);
    
    console.log('\nPlatform Distribution:');
    Object.entries(stats.platforms).forEach(([platform, count]) => {
      const percentage = ((count / stats.total) * 100).toFixed(1);
      console.log(`  ${platform.toUpperCase()}: ${count} streams (${percentage}%)`);
    });
    
    return stats;
    
  } catch (error) {
    console.error('Error getting platform stats:', error.message);
  }
}
```
