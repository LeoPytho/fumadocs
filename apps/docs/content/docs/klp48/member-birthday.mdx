## What is Birthday API?

The Birthday API from @jkt48/core allows you to retrieve birthday information for JKT48 members. This endpoint provides member names, birthdates, profile images, and related data in a streamlined format perfect for birthday calendars, notifications, and celebratory features.

- Member names and birthdates
- Profile images for birthday displays
- SHOWROOM room IDs for direct access
- URL-friendly identifiers
- Chronological birthday listings

<Callout>
This API endpoint returns birthday data for JKT48 members, making it easy to create birthday calendars, send birthday notifications, or display upcoming celebrations.

All features require an API key for authentication and access.
</Callout>

## Advanced Examples

### Today's Birthdays

```javascript
const jkt48Api = require('@jkt48/core');

async function getTodaysBirthdays() {
  try {
    const apiKey = 'your-api-key-here';
    const birthdays = await jkt48Api.birthday(apiKey);
    
    const today = new Date();
    const todayMonth = today.getMonth();
    const todayDate = today.getDate();
    
    const todaysBirthdays = birthdays.filter(member => {
      const birthDate = new Date(member.birthdate);
      return birthDate.getMonth() === todayMonth && birthDate.getDate() === todayDate;
    });
    
    if (todaysBirthdays.length > 0) {
      console.log("🎉 Today's Birthdays:");
      todaysBirthdays.forEach(member => {
        const age = today.getFullYear() - new Date(member.birthdate).getFullYear();
        console.log(`🎂 ${member.name} turns ${age} today!`);
      });
    } else {
      console.log("No birthdays today.");
    }
    
    return todaysBirthdays;
  } catch (error) {
    console.error('Error fetching today\'s birthdays:', error.message);
  }
}
```

### Upcoming Birthdays (Next 7 Days)

```javascript
async function getUpcomingBirthdays() {
  try {
    const apiKey = 'your-api-key-here';
    const birthdays = await jkt48Api.birthday(apiKey);
    
    const today = new Date();
    const nextWeek = new Date();
    nextWeek.setDate(today.getDate() + 7);
    
    const upcomingBirthdays = birthdays.filter(member => {
      const birthDate = new Date(member.birthdate);
      const thisYearBirthday = new Date(today.getFullYear(), birthDate.getMonth(), birthDate.getDate());
      
      // If birthday already passed this year, check next year
      if (thisYearBirthday < today) {
        thisYearBirthday.setFullYear(today.getFullYear() + 1);
      }
      
      return thisYearBirthday >= today && thisYearBirthday <= nextWeek;
    }).sort((a, b) => {
      const aDate = new Date(a.birthdate);
      const bDate = new Date(b.birthdate);
      const aThisYear = new Date(today.getFullYear(), aDate.getMonth(), aDate.getDate());
      const bThisYear = new Date(today.getFullYear(), bDate.getMonth(), bDate.getDate());
      
      return aThisYear - bThisYear;
    });
    
    console.log("📅 Upcoming Birthdays (Next 7 Days):");
    upcomingBirthdays.forEach(member => {
      const birthDate = new Date(member.birthdate);
      const thisYearBirthday = new Date(today.getFullYear(), birthDate.getMonth(), birthDate.getDate());
      const daysUntil = Math.ceil((thisYearBirthday - today) / (1000 * 60 * 60 * 24));
      
      console.log(`🎈 ${member.name} - ${daysUntil} day(s) to go`);
    });
    
    return upcomingBirthdays;
  } catch (error) {
    console.error('Error fetching upcoming birthdays:', error.message);
  }
}
```

### Birthday Calendar by Month

```javascript
async function getBirthdayCalendar() {
  try {
    const apiKey = 'your-api-key-here';
    const birthdays = await jkt48Api.birthday(apiKey);
    
    const monthNames = [
      "January", "February", "March", "April", "May", "June",
      "July", "August", "September", "October", "November", "December"
    ];
    
    const calendar = {};
    
    birthdays.forEach(member => {
      const birthDate = new Date(member.birthdate);
      const month = birthDate.getMonth();
      const day = birthDate.getDate();
      const monthName = monthNames[month];
      
      if (!calendar[monthName]) {
        calendar[monthName] = {};
      }
      
      if (!calendar[monthName][day]) {
        calendar[monthName][day] = [];
      }
      
      calendar[monthName][day].push({
        name: member.name,
        year: birthDate.getFullYear(),
        img: member.img,
        url_key: member.url_key
      });
    });
    
    // Display calendar
    Object.keys(calendar).forEach(month => {
      console.log(`\n📅 ${month.toUpperCase()}`);
      console.log('='.repeat(30));
      
      Object.keys(calendar[month])
        .sort((a, b) => parseInt(a) - parseInt(b))
        .forEach(day => {
          console.log(`${day.padStart(2, '0')}: ${calendar[month][day].map(m => `${m.name} (${m.year})`).join(', ')}`);
        });
    });
    
    return calendar;
  } catch (error) {
    console.error('Error creating birthday calendar:', error.message);
  }
}
```

### Birthday Age Calculator

```javascript
async function calculateMemberAges() {
  try {
    const apiKey = 'your-api-key-here';
    const birthdays = await jkt48Api.birthday(apiKey);
    
    const today = new Date();
    const membersWithAges = birthdays.map(member => {
      const birthDate = new Date(member.birthdate);
      let age = today.getFullYear() - birthDate.getFullYear();
      
      // Adjust if birthday hasn't occurred this year
      const monthDiff = today.getMonth() - birthDate.getMonth();
      if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birthDate.getDate())) {
        age--;
      }
      
      return {
        ...member,
        age,
        nextBirthdayIn: calculateDaysUntilNextBirthday(birthDate, today)
      };
    });
    
    // Sort by age (youngest to oldest)
    membersWithAges.sort((a, b) => a.age - b.age);
    
    console.log("👥 Members by Age:");
    membersWithAges.forEach(member => {
      console.log(`${member.name}: ${member.age} years old (next birthday in ${member.nextBirthdayIn} days)`);
    });
    
    return membersWithAges;
  } catch (error) {
    console.error('Error calculating member ages:', error.message);
  }
}

function calculateDaysUntilNextBirthday(birthDate, today) {
  const thisYearBirthday = new Date(today.getFullYear(), birthDate.getMonth(), birthDate.getDate());
  
  if (thisYearBirthday < today) {
    thisYearBirthday.setFullYear(today.getFullYear() + 1);
  }
  
  const diffTime = thisYearBirthday - today;
  return Math.ceil(diffTime / (1000 * 60 * 60 * 24));
}
```

### Birthday Notification System

```javascript
async function createBirthdayNotifications() {
  try {
    const apiKey = 'your-api-key-here';
    const birthdays = await jkt48Api.birthday(apiKey);
    
    const notifications = [];
    const today = new Date();
    
    birthdays.forEach(member => {
      const birthDate = new Date(member.birthdate);
      const daysUntil = calculateDaysUntilNextBirthday(birthDate, today);
      
      // Create notifications for different time periods
      if (daysUntil === 0) {
        notifications.push({
          type: 'today',
          priority: 'high',
          message: `🎉 It's ${member.name}'s birthday today!`,
          member: member,
          image: member.img
        });
      } else if (daysUntil === 1) {
        notifications.push({
          type: 'tomorrow',
          priority: 'medium',
          message: `🎂 ${member.name}'s birthday is tomorrow!`,
          member: member,
          image: member.img
        });
      } else if (daysUntil <= 7) {
        notifications.push({
          type: 'upcoming',
          priority: 'low',
          message: `📅 ${member.name}'s birthday is in ${daysUntil} days`,
          member: member,
          image: member.img
        });
      }
    });
    
    // Sort by priority
    const priorityOrder = { 'high': 3, 'medium': 2, 'low': 1 };
    notifications.sort((a, b) => priorityOrder[b.priority] - priorityOrder[a.priority]);
    
    console.log("🔔 Birthday Notifications:");
    notifications.forEach(notification => {
      console.log(`[${notification.priority.toUpperCase()}] ${notification.message}`);
    });
    
    return notifications;
  } catch (error) {
    console.error('Error creating birthday notifications:', error.message);
  }
}
```
