## Endpoint Details

<Tabs items={['All Members', 'Single Member', 'Response Format']}>

<Tab value="All Members">
**Get All Members**

**Base URL:** `https://v2.jkt48connect.com`  
**Endpoint:** `/api/klp48/members`  
**Method:** `GET`  
**Authentication:** API Key required

**Query Parameters:**
- `apikey` (required): Your API authentication key

**Example:**
```http
GET /api/klp48/members?apikey=YOUR_API_KEY HTTP/1.1
Host: v2.jkt48connect.com
```
</Tab>

<Tab value="Single Member">
**Get Member by Name**

**Endpoint:** `/api/klp48/member/:name`  
**Method:** `GET`  
**Authentication:** API Key required

**Parameters:**
- `name` (required): Member's full name or nickname
- `apikey` (required): Your API authentication key

**Examples:**
```http
GET /api/klp48/member/hillary?apikey=YOUR_API_KEY HTTP/1.1
GET /api/klp48/member/Shuen Hiu Yau?apikey=YOUR_API_KEY HTTP/1.1
GET /api/klp48/member/Salwa?apikey=YOUR_API_KEY HTTP/1.1
Host: v2.jkt48connect.com
```
</Tab>

import requests
import urllib.parse
from typing import List, Dict, Optional

API_KEY = 'YOUR_API_KEY'
BASE_URL = 'https://v2.jkt48connect.com'

def get_all_klp48_members() -> List[Dict]:
    """Fetch all KLP48 members"""
    url = f"{BASE_URL}/api/klp48/members"
    params = {'apikey': API_KEY}
    
    try:
        response = requests.get(url, params=params)
        response.raise_for_status()
        return response.json()
    except requests.exceptions.RequestException as e:
        print(f"Error fetching KLP48 members: {e}")
        raise

def get_klp48_member_by_name(name: str) -> Optional[Dict]:
    """Get specific KLP48 member by name or nickname"""
    encoded_name = urllib.parse.quote(name)
    url = f"{BASE_URL}/api/klp48/member/{encoded_name}"
    params = {'apikey': API_KEY}
    
    try:
        response = requests.get(url, params=params)
        response.raise_for_status()
        return response.json()
    except requests.exceptions.RequestException as e:
        print(f"Error fetching member {name}: {e}")
        return None

def filter_members_by_birthplace(members: List[Dict], birthplace: str) -> List[Dict]:
    """Filter members by birthplace"""
    return [member for member in members 
            if member.get('birthplace') and birthplace.lower() in member['birthplace'].lower()]

def get_member_age_range(members: List[Dict], min_year: int, max_year: int) -> List[Dict]:
    """Get members born within specific year range"""
    filtered = []
    for member in members:
        if member.get('birthday'):
            try:
                birth_year = int(member['birthday'].split('.')[-1])
                if min_year <= birth_year <= max_year:
                    filtered.append(member)
            except ValueError:
                continue
    return filtered

def display_member_profile(member: Dict):
    """Display formatted member profile"""
    print(f"=== {member['name']} ===")
    print(f"Nickname: {', '.join(member['nicknames'])}")
    print(f"Birthday: {member.get('birthday', 'N/A')}")
    print(f"Birthplace: {member.get('birthplace', 'N/A')}")
    print(f"Height: {member.get('height', 'N/A')}")
    print(f"Charm Point: {member.get('charm_point', 'N/A')}")
    print(f"Special Skills: {member.get('special_skills', 'N/A')}")
    print(f"Hobby: {member.get('hobby', 'N/A')}")
    print(f"Social Media: {len(member.get('socials', []))} platforms")
    print("-" * 50)

# Usage example
if __name__ == "__main__":
    try:
        # Get all members
        members = get_all_klp48_members()
        print(f"Total KLP48 Members: {len(members)}")
        
        # Get specific member
        hillary = get_klp48_member_by_name('hillary')
        if hillary:
            display_member_profile(hillary)
        
        # Filter by country
        japanese_members = filter_members_by_birthplace(members, 'Japan')
        malaysian_members = filter_members_by_birthplace(members, 'Malaysia')
        indonesian_members = filter_members_by_birthplace(members, 'Indonesia')
        
        print(f"Japanese members: {len(japanese_members)}")
        print(f"Malaysian members: {len(malaysian_members)}")
        print(f"Indonesian members: {len(indonesian_members)}")
        
        # Get members born in 2000s
        gen_2000s = get_member_age_range(members, 2000, 2009)

package main

import (
    "encoding/json"
    "fmt"
    "net/http"
    "net/url"
    "strings"
    "time"
)

const (
    APIKey  = "YOUR_API_KEY"
    BaseURL = "https://v2.jkt48connect.com"
)

type Social struct {
    Title string `json:"title"`
    URL   string `json:"url"`
}

type KLP48Member struct {
    ID           string   `json:"_id"`
    Name         string   `json:"name"`
    Nicknames    []string `json:"nicknames"`
    Image        string   `json:"img"`
    ImageAlt     string   `json:"img_alt"`
    URL          string   `json:"url"`
    Group        string   `json:"group"`
    Socials      []Social `json:"socials"`
    RoomID       *int     `json:"room_id"`
    SRExists     bool     `json:"sr_exists"`
    IsGraduate   bool     `json:"is_graduate"`
    Generation   string   `json:"generation"`
    IDNUsername  *string  `json:"idn_username"`
    Birthday     string   `json:"birthday"`
    Birthplace   string   `json:"birthplace"`
    Height       string   `json:"height"`
    CharmPoint   string   `json:"charm_point"`
    SpecialSkills string  `json:"special_skills"`
    Hobby        string   `json:"hobby"`
}

func getAllKLP48Members() ([]KLP48Member, error) {
    url := fmt.Sprintf("%s/api/klp48/members?apikey=%s", BaseURL, APIKey)
    
    client := &http.Client{Timeout: 30 * time.Second}
    resp, err := client.Get(url)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    if resp.StatusCode != http.StatusOK {
        return nil, fmt.Errorf("API request failed with status: %d", resp.StatusCode)
    }
    
    var members []KLP48Member
    err = json.NewDecoder(resp.Body).Decode(&members)
    return members, err
}

func getKLP48MemberByName(name string) (*KLP48Member, error) {
    encodedName := url.QueryEscape(name)
    url := fmt.Sprintf("%s/api/klp48/member/%s?apikey=%s", BaseURL, encodedName, APIKey)
    
    client := &http.Client{Timeout: 30 * time.Second}
    resp, err := client.Get(url)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    if resp.StatusCode != http.StatusOK {
        return nil, fmt.Errorf("API request failed with status: %d", resp.StatusCode)
    }
    
    var member KLP48Member
    err = json.NewDecoder(resp.Body).Decode(&member)
    if err != nil {
        return nil, err
    }
    
    return &member, nil
}

func filterMembersByBirthplace(members []KLP48Member, birthplace string) []KLP48Member {
    var filtered []KLP48Member
    lowerBirthplace := strings.ToLower(birthplace)
    
    for _, member := range members {
        if strings.Contains(strings.ToLower(member.Birthplace), lowerBirthplace) {
            filtered = append(filtered, member)
        }
    }
    return filtered
}

func displayMemberProfile(member KLP48Member) {

func main() {

    fmt.Printf("Japanese members: %d\n", len(japaneseMembers))
    fmt.Printf("Malaysian members: %d\n", len(malaysianMembers))

## Data Structure

<Callout type="info">
Each KLP48 member object contains comprehensive profile information including personal details, social media links, and physical characteristics.
</Callout>

**Key Fields:**

| Field | Type | Description |
|-------|------|-------------|
| `_id` | string | Unique member identifier |
| `name` | string | Full member name (may include Chinese characters) |
| `nicknames` | array | List of member nicknames |
| `img` | string | Primary profile image URL |
| `img_alt` | string | Alternative profile image URL |
| `url` | string | Member's profile URL slug |
| `group` | string | Group affiliation (klp48) |
| `socials` | array | Social media platform links (Instagram, X) |
| `birthday` | string | Birthday in DD.MM.YYYY format |
| `birthplace` | string | Birthplace with country |
| `height` | string | Height in centimeters |
| `charm_point` | string | Member's charm points |
| `special_skills` | string | Special skills and talents |
| `hobby` | string | Hobbies and interests |
| `generation` | string | Member's generation (gen1-klp48) |

// Demographic analysis
function analyzeKLP48Demographics(members) {
  const countries = {};
  const ageGroups = {
    '1999-2000': 0,
    '2001-2003': 0,
    '2004-2006': 0,
    '2007-2009': 0
  };
  
  members.forEach(member => {
    // Count by country
    const country = member.birthplace.split(',').pop().trim();
    countries[country] = (countries[country] || 0) + 1;
    
    // Age groups by birth year
    const year = parseInt(member.birthday.split('.')[2]);
    if (year >= 1999 && year <= 2000) ageGroups['1999-2000']++;
    else if (year >= 2001 && year <= 2003) ageGroups['2001-2003']++;
    else if (year >= 2004 && year <= 2006) ageGroups['2004-2006']++;
    else if (year >= 2007 && year <= 2009) ageGroups['2007-2009']++;
  });
  
  return { countries, ageGroups };
}

async function safelyFetchKLP48Member(name) {
  try {
    const member = await getKLP48MemberByName(name);
    
    if (!member) {

      return null;

      return null;
