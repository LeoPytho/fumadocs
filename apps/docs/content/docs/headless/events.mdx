## What is Events API?

The Events API from @jkt48/core allows you to retrieve comprehensive information about JKT48 events and activities. This endpoint provides detailed event data including dates, titles, categories, and direct links to official event pages, making it perfect for event calendars, notifications, and fan applications.

- Complete event listings with dates and titles
- Event categories with visual indicators
- Direct links to official JKT48 event pages
- Unique event identifiers for tracking
- Chronological event organization

<Callout>
This API endpoint returns data for all JKT48 events, providing a comprehensive overview of upcoming performances, festivals, and activities.

All features require an API key for authentication and access.
</Callout>

## Advanced Examples

### Upcoming Events (Next 30 Days)

```javascript
const jkt48Api = require('@jkt48/core');

async function getUpcomingEvents() {
  try {
    const apiKey = 'your-api-key-here';
    const events = await jkt48Api.events(apiKey);
    
    const today = new Date();
    const next30Days = new Date();
    next30Days.setDate(today.getDate() + 30);
    
    const upcomingEvents = events.filter(event => {
      const eventDate = new Date(event.date);
      return eventDate >= today && eventDate <= next30Days;
    }).sort((a, b) => new Date(a.date) - new Date(b.date));
    
    console.log("📅 Upcoming Events (Next 30 Days):");
    upcomingEvents.forEach(event => {
      const eventDate = new Date(event.date);
      const daysUntil = Math.ceil((eventDate - today) / (1000 * 60 * 60 * 24));
      
      console.log(`🎪 ${event.title}`);
      console.log(`   📅 ${eventDate.toLocaleDateString()} (${daysUntil} days to go)`);
      console.log(`   🔗 ${event.url}\n`);
    });
    
    return upcomingEvents;
  } catch (error) {
    console.error('Error fetching upcoming events:', error.message);
  }
}
```

### Events by Month

```javascript
async function getEventsByMonth() {
  try {
    const apiKey = 'your-api-key-here';
    const events = await jkt48Api.events(apiKey);
    
    const monthNames = [
      "January", "February", "March", "April", "May", "June",
      "July", "August", "September", "October", "November", "December"
    ];
    
    const eventsByMonth = {};
    
    events.forEach(event => {
      const eventDate = new Date(event.date);
      const monthYear = `${monthNames[eventDate.getMonth()]} ${eventDate.getFullYear()}`;
      
      if (!eventsByMonth[monthYear]) {
        eventsByMonth[monthYear] = [];
      }
      
      eventsByMonth[monthYear].push(event);
    });
    
    // Display events by month
    Object.keys(eventsByMonth)
      .sort((a, b) => new Date(a) - new Date(b))
      .forEach(monthYear => {
        console.log(`\n📅 ${monthYear.toUpperCase()}`);
        console.log('='.repeat(40));
        
        eventsByMonth[monthYear]
          .sort((a, b) => new Date(a.date) - new Date(b.date))
          .forEach(event => {
            const eventDate = new Date(event.date);
            console.log(`${eventDate.getDate().toString().padStart(2, '0')}: ${event.title}`);
          });
      });
    
    return eventsByMonth;
  } catch (error) {
    console.error('Error organizing events by month:', error.message);
  }
}
```

### Event Search and Filter

```javascript
async function searchEvents(searchTerm) {
  try {
    const apiKey = 'your-api-key-here';
    const events = await jkt48Api.events(apiKey);
    
    const filteredEvents = events.filter(event => 
      event.title.toLowerCase().includes(searchTerm.toLowerCase()) ||
      event.id.toLowerCase().includes(searchTerm.toLowerCase())
    );
    
    console.log(`🔍 Search Results for "${searchTerm}":`);
    
    if (filteredEvents.length === 0) {
      console.log("No events found matching your search.");
      return [];
    }
    
    filteredEvents.forEach(event => {

      console.log(`\n🎪 ${event.title}`);
      console.log(`   ID: ${event.id}`);
      console.log(`   📅 Date: ${eventDate.toLocaleDateString()}`);
      console.log(`   🔗 URL: ${event.url}`);
    });
    
    return filteredEvents;
  } catch (error) {
    console.error('Error searching events:', error.message);
  }
}
```

### Event Countdown Timer

```javascript
async function createEventCountdowns() {
  try {
    const apiKey = 'your-api-key-here';
    const events = await jkt48Api.events(apiKey);
    
    const today = new Date();
    const futureEvents = events.filter(event => new Date(event.date) > today)
      .sort((a, b) => new Date(a.date) - new Date(b.date))
      .slice(0, 5); // Get next 5 events
    
    console.log("⏰ Event Countdowns:");
    console.log("=".repeat(50));
    
    futureEvents.forEach(event => {
      const eventDate = new Date(event.date);
      const timeUntil = eventDate - today;
      
      const days = Math.floor(timeUntil / (1000 * 60 * 60 * 24));
      const hours = Math.floor((timeUntil % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
      const minutes = Math.floor((timeUntil % (1000 * 60 * 60)) / (1000 * 60));
      
      console.log(`\n🎪 ${event.title}`);
      console.log(`⏱️  ${days} days, ${hours} hours, ${minutes} minutes`);
      console.log(`📅 ${eventDate.toLocaleString()}`);
    });
    
    return futureEvents;
  } catch (error) {
    console.error('Error creating event countdowns:', error.message);
  }
}
```

### Event Calendar Generator

```javascript
async function generateEventCalendar() {
  try {
    const apiKey = 'your-api-key-here';
    const events = await jkt48Api.events(apiKey);
    
    const calendar = {
      today: [],
      thisWeek: [],
      thisMonth: [],
      upcoming: []
    };
    
    const today = new Date();
    const startOfWeek = new Date(today);
    startOfWeek.setDate(today.getDate() - today.getDay());
    const endOfWeek = new Date(startOfWeek);
    endOfWeek.setDate(startOfWeek.getDate() + 6);
    
    const startOfMonth = new Date(today.getFullYear(), today.getMonth(), 1);
    const endOfMonth = new Date(today.getFullYear(), today.getMonth() + 1, 0);
    
    events.forEach(event => {
      const eventDate = new Date(event.date);
      
      // Check if event is today
      if (eventDate.toDateString() === today.toDateString()) {
        calendar.today.push(event);
      }
      // Check if event is this week
      else if (eventDate >= startOfWeek && eventDate <= endOfWeek) {
        calendar.thisWeek.push(event);
      }
      // Check if event is this month
      else if (eventDate >= startOfMonth && eventDate <= endOfMonth) {
        calendar.thisMonth.push(event);
      }
      // Future events
      else if (eventDate > endOfMonth) {
        calendar.upcoming.push(event);
      }
    });
    
    // Display calendar
    console.log("📅 JKT48 EVENT CALENDAR");
    console.log("=".repeat(50));
    
    if (calendar.today.length > 0) {
      console.log("\n🔥 TODAY'S EVENTS:");
      calendar.today.forEach(event => {
        console.log(`   • ${event.title}`);
      });
    }
    
    if (calendar.thisWeek.length > 0) {
      console.log("\n📋 THIS WEEK:");
      calendar.thisWeek.forEach(event => {
        const eventDate = new Date(event.date);
        console.log(`   • ${eventDate.toLocaleDateString()} - ${event.title}`);
      });
    }
    
    if (calendar.thisMonth.length > 0) {
      console.log("\n📆 THIS MONTH:");
      calendar.thisMonth.forEach(event => {
        const eventDate = new Date(event.date);
        console.log(`   • ${eventDate.toLocaleDateString()} - ${event.title}`);
      });
    }
    
    if (calendar.upcoming.length > 0) {
      console.log("\n🔮 UPCOMING:");
      calendar.upcoming.slice(0, 10).forEach(event => {
        const eventDate = new Date(event.date);
        console.log(`   • ${eventDate.toLocaleDateString()} - ${event.title}`);
      });
    }
    
    return calendar;
  } catch (error) {
    console.error('Error generating event calendar:', error.message);
  }
}
```

### Event Notification System

```javascript
async function createEventNotifications() {
  try {
    const apiKey = 'your-api-key-here';
    const events = await jkt48Api.events(apiKey);
    
    const notifications = [];
    const today = new Date();
    
    events.forEach(event => {
      const eventDate = new Date(event.date);
      const daysUntil = Math.ceil((eventDate - today) / (1000 * 60 * 60 * 24));
      
      // Create notifications for different time periods
      if (daysUntil === 0) {
        notifications.push({
          type: 'today',
          priority: 'high',
          message: `🔴 LIVE NOW: ${event.title}`,
          event: event,
          urgency: 'immediate'
        });
      } else if (daysUntil === 1) {
        notifications.push({
          type: 'tomorrow',
          priority: 'high',
          message: `🟡 TOMORROW: ${event.title}`,
          event: event,
          urgency: 'high'
        });
      } else if (daysUntil <= 7) {
        notifications.push({
          type: 'thisWeek',
          priority: 'medium',
          message: `🟢 In ${daysUntil} days: ${event.title}`,
          event: event,
          urgency: 'medium'
        });
      } else if (daysUntil <= 30) {
        notifications.push({
          type: 'upcoming',
          priority: 'low',
          message: `📅 ${event.title} - ${eventDate.toLocaleDateString()}`,
          event: event,
          urgency: 'low'
        });
      }
    });
    
    // Sort by priority and date
    const priorityOrder = { 'high': 3, 'medium': 2, 'low': 1 };
    notifications.sort((a, b) => {
      if (a.priority !== b.priority) {
        return priorityOrder[b.priority] - priorityOrder[a.priority];
      }
      return new Date(a.event.date) - new Date(b.event.date);
    });
    
    console.log("🔔 EVENT NOTIFICATIONS:");
    console.log("=".repeat(40));
    
    notifications.forEach(notification => {
      console.log(`[${notification.urgency.toUpperCase()}] ${notification.message}`);
    });
    
    return notifications;
  } catch (error) {
    console.error('Error creating event notifications:', error.message);
  }
}
```
