## What is Recent Live API?

The Recent Live API from @jkt48/core allows you to retrieve comprehensive information about recent JKT48 member live streaming sessions. This endpoint provides detailed live stream data including member information, viewer statistics, gift totals, and platform details in a single API call.

- Complete live stream records from SHOWROOM and IDN platforms
- Member profile information and social media data
- Live stream duration and viewer statistics
- Gift totals and monetization data
- Platform-specific metadata (IDN Live and SHOWROOM)
- Real-time streaming status tracking

<Callout>
This API endpoint returns data for recent live streaming sessions from all JKT48 members across multiple platforms, providing comprehensive streaming analytics.

All features require an API key for authentication and access.
</Callout>

## Data Field Explanations

<Accordions type="single">

<Accordion title="Basic Information">
- **_id**: Unique identifier for the live stream record
- **data_id**: Platform-specific data identifier
- **type**: Streaming platform ("idn" or "showroom")
- **created_at**: Timestamp when the live stream ended
</Accordion>

<Accordion title="Member Information">
- **member.name**: Full name of the streaming member
- **member.nickname**: Member's nickname or stage name
- **member.img**: Primary profile image URL
- **member.img_alt**: Alternative high-quality profile image
- **member.url**: URL-friendly member identifier
- **member.is_graduate**: Whether the member has graduated
- **member.is_official**: Whether this is an official JKT48 account
</Accordion>

<Accordion title="Live Stream Data">
- **live_info.duration**: Stream duration in milliseconds
- **live_info.viewers.num**: Peak viewer count
- **live_info.viewers.is_excitement**: Whether the stream reached excitement mode
- **live_info.date.start**: Stream start timestamp
- **live_info.date.end**: Stream end timestamp
</Accordion>

<Accordion title="Platform Specific">
- **idn**: IDN Live platform data (present for IDN streams)
- **room_id**: Platform room identifier
- **gift_rate**: Gift conversion rate
- **points**: Points earned from gifts
- **total_gift**: Total gift amount in Indonesian Rupiah
</Accordion>

</Accordions>

## Advanced Examples

### Processing Recent Live Data

```javascript
const jkt48Api = require('@jkt48/core');

async function processRecentLives() {
  try {
    const apiKey = 'your-api-key-here';
    const recent = await jkt48Api.recent(apiKey);
    
    // Separate by platform
    const idnStreams = recent.filter(stream => stream.type === 'idn');
    const showroomStreams = recent.filter(stream => stream.type === 'showroom');
    
    console.log(`Total streams: ${recent.length}`);
    console.log(`IDN streams: ${idnStreams.length}`);
    console.log(`SHOWROOM streams: ${showroomStreams.length}`);
    
    // Calculate total earnings
    const totalEarnings = recent.reduce((sum, stream) => {
      const amount = parseInt(stream.total_gift.replace(/[^\d]/g, ''));
      return sum + amount;
    }, 0);
    
    console.log(`Total earnings: Rp ${totalEarnings.toLocaleString('id-ID')}`);
  } catch (error) {
    console.error('Error fetching recent lives:', error.message);
  }
}
```

### Get Top Performing Streams

```javascript
async function getTopStreams() {
  try {
    const apiKey = 'your-api-key-here';
    const recent = await jkt48Api.recent(apiKey);
    
    // Sort by viewer count
    const topByViewers = recent
      .sort((a, b) => b.live_info.viewers.num - a.live_info.viewers.num)
      .slice(0, 5);
    
    console.log('Top 5 streams by viewers:');
    topByViewers.forEach((stream, index) => {
      console.log(`${index + 1}. ${stream.member.nickname} - ${stream.live_info.viewers.num} viewers`);
    });
    
    // Sort by gift earnings
    const topByGifts = recent
      .sort((a, b) => {
        const aAmount = parseInt(a.total_gift.replace(/[^\d]/g, ''));
        const bAmount = parseInt(b.total_gift.replace(/[^\d]/g, ''));
        return bAmount - aAmount;
      })
      .slice(0, 5);
    
    console.log('\nTop 5 streams by earnings:');
    topByGifts.forEach((stream, index) => {
      console.log(`${index + 1}. ${stream.member.nickname} - ${stream.total_gift}`);
    });
  } catch (error) {
    console.error('Error processing top streams:', error.message);
  }
}
```

### Filter by Member

```javascript
async function getMemberStreams(memberName) {
  try {
    const apiKey = 'your-api-key-here';
    const recent = await jkt48Api.recent(apiKey);
    
    const memberStreams = recent.filter(stream => 
      stream.member.name.toLowerCase().includes(memberName.toLowerCase()) ||
      (stream.member.nickname && stream.member.nickname.toLowerCase().includes(memberName.toLowerCase()))
    );
    
    if (memberStreams.length > 0) {
      console.log(`Found ${memberStreams.length} recent streams for ${memberName}:`);
      memberStreams.forEach(stream => {
        const duration = Math.floor(stream.live_info.duration / 60000); // Convert to minutes
        console.log(`- ${stream.live_info.date.start}: ${duration} mins, ${stream.live_info.viewers.num} viewers, ${stream.total_gift}`);
      });
    } else {
      console.log(`No recent streams found for ${memberName}`);
    }
  } catch (error) {
    console.error('Error searching member streams:', error.message);
  }
}

// Usage
getMemberStreams('Intan');
```
